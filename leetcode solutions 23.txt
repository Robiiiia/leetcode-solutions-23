136. single tree

import java.util.HashMap; // Импортируем HashMap, чтобы хранить числа и количество их появлений.
import java.util.Map;     // Импортируем Map для работы с парами ключ-значение.

class Solution {

    public int singleNumber(int[] nums) {

        // Создаём HashMap:
        // Integer — тип ключа, то есть самого числа.
        // Integer — тип значения, то есть количества появлений этого числа.
        Map<Integer, Integer> map = new HashMap<>();

        // Перебираем все числа массива nums.
        for (int num : nums) {

            // Проверяем, есть ли текущее число num в HashMap.
            if (map.containsKey(num)) {

                // Если число уже есть, увеличиваем его количество на 1.
                map.put(num, map.get(num) + 1);

            } else {

                // Если числа ещё нет, добавляем его в HashMap
                // и устанавливаем количество появлений равным 1.
                map.put(num, 1);
            }
        }

        // Перебираем все числа массива ещё раз.
        for (int num : nums) {

            // Получаем количество появлений числа num.
            // Если оно равно 1, значит, нашли искомое число.
            if (map.get(num) == 1) {

                // Возвращаем число, которое встретилось только один раз.
                return num;
            }
        }

        // Эта строка нужна, чтобы метод имел возвращаемое значение
        // даже в случае, если подходящее число не найдено.
        // По условию задачи такого случая быть не должно.
        return -1;
    }
}
